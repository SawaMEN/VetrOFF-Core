// One gate for explicit diagnostics across all selectors in this process.
var adaptiveDiagnosticTester ac.SiberianTester

// AdaptiveDNSResolver is a locally supplied DNS-over-TLS endpoint. It is
// never taken from a subscription or resolved via the system resolver.
type AdaptiveDNSResolver struct {
	IP netip.Addr
	ServerName string
}

func (r AdaptiveDNSResolver) valid() bool {
	if !ac.PublicDiagnosticIP(r.IP)||r.IP.Is4In6()||r.ServerName!=strings.ToLower(r.ServerName)||
		!strings.Contains(r.ServerName,".")||net.ParseIP(r.ServerName)!=nil||
		len(r.ServerName)>253||strings.HasSuffix(r.ServerName,".") {return false}
	for _,label:=range strings.Split(r.ServerName,".") {
		if len(label)==0||len(label)>63||label[0]=='-'||label[len(label)-1]=='-' {return false}
		for _,c:=range label {if !((c>='a'&&c<='z')||(c>='0'&&c<='9')||c=='-') {return false}}
	}
	return true
}

type adaptiveTunnelPath struct {
	owner *AdaptiveSelectorOwner
	epoch *adaptiveEpoch
	outbound adapter.Outbound
	revision uint64
	resolver AdaptiveDNSResolver
	physical *adaptivePhysicalBinding
}

func (p *adaptiveTunnelPath) Scope() ac.DiagnosticScope {
	if p.physical!=nil {return ac.ScopePhysical}
	return ac.ScopeTunnel
}

func (p *adaptiveTunnelPath) Generation() uint64 {
	if p.epoch.ctx.Err()!=nil||p.owner.epoch.Load()!=p.epoch {return 0}
	if p.physical!=nil&&!p.physical.valid() {return 0}
	s:=p.owner.binding.selector
	s.adaptiveSelectionMu.Lock()
	defer s.adaptiveSelectionMu.Unlock()
	if s.adaptiveSelectionRevision!=p.revision||s.selected.Load()!=p.outbound {return 0}
	return 1
}

func (p *adaptiveTunnelPath) dial(ctx context.Context,ip netip.Addr,port string)(net.Conn,error) {
	if p.Generation()==0 {return nil,errors.New("diagnostic path changed")}
	if !ac.PublicDiagnosticIP(ip) {return nil,errors.New("non-public diagnostic address")}
	if err:=ctx.Err();err!=nil {return nil,err}
	var conn net.Conn
	var err error
	if p.physical!=nil {
		conn,err=p.physical.dial(ctx,ip,port)
	} else {
		conn,err=p.outbound.DialContext(ctx,"tcp",M.ParseSocksaddrHostPortStr(ip.String(),port))
	}
	if err!=nil||conn==nil {
		if conn!=nil {_=conn.Close()}
		if err==nil {err=errors.New("outbound returned no connection")}
		return nil,err
	}
	if ctx.Err()!=nil||p.Generation()==0 {
		_=conn.Close()
		return nil,errors.New("diagnostic path changed")
	}
	return conn,nil
}

func (p *adaptiveTunnelPath) DialContext(ctx context.Context,network,address string)(net.Conn,error) {
	host,port,err:=net.SplitHostPort(address)
	if err!=nil||network!="tcp"||port!="443" {return nil,errors.New("invalid diagnostic destination")}
	ip,err:=netip.ParseAddr(host)
	if err!=nil {return nil,errors.New("diagnostic destination must be a pinned IP")}
	return p.dial(ctx,ip,port)
}

// Query exactly the requested FQDN through the pinned leaf via authenticated
// DNS-over-TLS. No search suffix, hosts file, system DNS or DNS fallback.
func (p *adaptiveTunnelPath) LookupNetIP(ctx context.Context,network,host string)([]netip.Addr,error) {
	if !strings.HasSuffix(host,".")||len(host)>254 {return nil,errors.New("expected absolute DNS name")}
	var types []uint16
	switch network {
	case "ip4":types=[]uint16{dns.TypeA}
	case "ip6":types=[]uint16{dns.TypeAAAA}
	case "ip":types=[]uint16{dns.TypeA,dns.TypeAAAA}
	default:return nil,errors.New("unsupported DNS family")
	}
	var result []netip.Addr
	for _,typ:=range types {
		ips,err:=p.query(ctx,host,typ)
		if err!=nil {return nil,err}
		result=append(result,ips...)
	}
	return result,nil
}

func (p *adaptiveTunnelPath) query(ctx context.Context,host string,typ uint16)([]netip.Addr,error) {
	ctx,cancel:=context.WithTimeout(ctx,3*time.Second)
	defer cancel()
	conn,err:=p.dial(ctx,p.resolver.IP,"853")
	if err!=nil {return nil,err}
	defer conn.Close()
	deadline,_:=ctx.Deadline()
	if err=conn.SetDeadline(deadline);err!=nil {return nil,err}
	stop:=context.AfterFunc(ctx,func(){_=conn.Close()})
	defer stop()
	tlsConn:=tls.Client(conn,&tls.Config{ServerName:p.resolver.ServerName,MinVersion:tls.VersionTLS12})
	if err=tlsConn.HandshakeContext(ctx);err!=nil {return nil,err}
	msg:=new(dns.Msg)
	msg.SetQuestion(host,typ)
	wire:=&dns.Conn{Conn:tlsConn}
	if err=wire.WriteMsg(msg);err!=nil {return nil,err}
	response,err:=wire.ReadMsg()
	if err!=nil {return nil,err}
	if ctx.Err()!=nil||p.Generation()==0 {return nil,errors.New("diagnostic path changed")}
	return adaptiveDNSAnswers(msg,response)
}

// Do not use glue or unrelated answer records. A CNAME chain must be
// complete within this response; no additional implicit queries are made.
func adaptiveDNSAnswers(request,response *dns.Msg)([]netip.Addr,error) {
	if request==nil||response==nil||len(request.Question)!=1||!response.Response||
		response.Id!=request.Id||response.Truncated||response.Opcode!=dns.OpcodeQuery||
		len(response.Question)!=1||len(response.Answer)>128 {
		return nil,errors.New("invalid diagnostic DNS response")
	}
	q,rq:=request.Question[0],response.Question[0]
	if !strings.EqualFold(q.Name,rq.Name)||q.Qtype!=rq.Qtype||q.Qclass!=rq.Qclass {
		return nil,errors.New("DNS question mismatch")
	}
	switch response.Rcode {
	case dns.RcodeNameError:return nil,&net.DNSError{Err:"name not found",Name:q.Name,IsNotFound:true}
	case dns.RcodeSuccess:
	default:return nil,errors.New("diagnostic DNS query failed")
	}
	name:=strings.ToLower(q.Name)
	seen:=map[string]bool{}
	for i:=0;i<8;i++ {
		if seen[name] {return nil,errors.New("DNS alias loop")}
		seen[name]=true
		var result []netip.Addr
		var alias string
		for _,rr:=range response.Answer {
			if !strings.EqualFold(rr.Header().Name,name)||rr.Header().Class!=dns.ClassINET {continue}
			var ip net.IP
			switch a:=rr.(type) {
			case *dns.CNAME:
				target:=strings.ToLower(a.Target)
				if alias!=""&&alias!=target {return nil,errors.New("conflicting DNS aliases")}
				alias=target
			case *dns.A:if q.Qtype==dns.TypeA {ip=a.A}
			case *dns.AAAA:if q.Qtype==dns.TypeAAAA {ip=a.AAAA}
			}
			if ip!=nil {
				addr,ok:=netip.AddrFromSlice(ip)
				if !ok {return nil,errors.New("invalid DNS address")}
				addr=addr.Unmap()
				if !ac.PublicDiagnosticIP(addr)||(q.Qtype==dns.TypeA)!=addr.Is4() {return nil,errors.New("unsafe DNS answer")}
				result=append(result,addr)
			}
		}
		if len(result)>32 {return nil,errors.New("too many DNS addresses")}
		if len(result)>0 {
			if alias!="" {return nil,errors.New("ambiguous DNS alias response")}
			return result,nil
		}
		if alias=="" {
			if i>0 {return nil,errors.New("incomplete DNS alias response")}
			return nil,nil
		}
		name=alias
	}
	return nil,errors.New("DNS alias limit exceeded")
}

// beginTunnelDiagnostic pins the current leaf and pauses automatic probe
// batches for the duration of a user-requested diagnostic. It never changes
// selection, reserves user budget, or records health. Network changes and
// manual selections cancel the epoch even if a diagnostic is running.
func (s *Selector) beginTunnelDiagnostic(caller context.Context,resolver AdaptiveDNSResolver,consent bool)(context.Context,*adaptiveTunnelPath,func(),error) {
	if !consent||!resolver.valid() {return nil,nil,nil,errors.New("local diagnostic consent and valid DNS-over-TLS endpoint required")}
	s.adaptiveLifecycleMu.Lock()
	defer s.adaptiveLifecycleMu.Unlock()
	o:=s.adaptiveOwner
	if o==nil {return nil,nil,nil,ac.ErrControllerInactive}
	o.mu.Lock()
	defer o.mu.Unlock()
	e:=o.epoch.Load()
	if o.closed||o.running||o.diagnosticRunning||e==nil||e.ctx.Err()!=nil||caller.Err()!=nil {
		return nil,nil,nil,ac.ErrControllerInactive
	}
	s.adaptiveSelectionMu.Lock()
	selected:=s.selected.Load()
	revision:=s.adaptiveSelectionRevision
	s.adaptiveSelectionMu.Unlock()
	if selected==nil||o.binding.members[selected.Tag()]!=selected {return nil,nil,nil,errors.New("unverified diagnostic leaf")}
	ctx,cancel:=context.WithCancel(e.ctx)
	stop:=context.AfterFunc(caller,cancel)
	o.diagnosticRunning=true
	p:=&adaptiveTunnelPath{owner:o,epoch:e,outbound:selected,revision:revision,resolver:resolver}
	finish:=func(){
		stop();cancel()
		o.mu.Lock()
		o.diagnosticRunning=false
		o.mu.Unlock()
	}
	return ctx,p,finish,nil
}

func (s *Selector) AdaptiveSiberian(ctx context.Context,resolver AdaptiveDNSResolver,req ac.SiberianRequest)(ac.SiberianReport,error) {
	ctx,p,finish,err:=s.beginTunnelDiagnostic(ctx,resolver,req.Consent)
	if err!=nil {return ac.SiberianReport{},err}
	defer finish()
	return adaptiveCheckSiberian(ctx,p,req)
}

func (s *Selector) AdaptiveConnectivity(ctx context.Context,resolver AdaptiveDNSResolver,req ac.ConnectivityRequest)(ac.ConnectivityReport,error) {
	ctx,p,finish,err:=s.beginTunnelDiagnostic(ctx,resolver,req.Consent)
	if err!=nil {return ac.ConnectivityReport{},err}
	defer finish()
	return adaptiveDiagnosticTester.CheckConnectivity(ctx,p,req)
}
