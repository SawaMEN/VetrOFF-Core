func TestAntiDPIDiagnosticDNSAnswer(t *testing.T) {
	q:=new(dns.Msg)
	q.SetQuestion("owned.example.",dns.TypeA)
	r:=new(dns.Msg)
	r.SetReply(q)
	r.Answer=[]dns.RR{
		&dns.CNAME{Hdr:dns.RR_Header{Name:"owned.example.",Rrtype:dns.TypeCNAME,Class:dns.ClassINET},Target:"alias.example."},
		&dns.A{Hdr:dns.RR_Header{Name:"alias.example.",Rrtype:dns.TypeA,Class:dns.ClassINET},A:net.ParseIP("1.1.1.1")},
		&dns.A{Hdr:dns.RR_Header{Name:"unrelated.example.",Rrtype:dns.TypeA,Class:dns.ClassINET},A:net.ParseIP("127.0.0.1")},
	}
	ips,err:=adaptiveDNSAnswers(q,r)
	if err!=nil||len(ips)!=1||ips[0]!=netip.MustParseAddr("1.1.1.1") {t.Fatal(ips,err)}
	r.Answer[1].(*dns.A).A=net.ParseIP("127.0.0.1")
	if _,err=adaptiveDNSAnswers(q,r);err==nil {t.Fatal("private IP accepted")}
	r.Id++
	if _,err=adaptiveDNSAnswers(q,r);err==nil {t.Fatal("mismatched response accepted")}
}

func TestAntiDPIDiagnosticPinnedEpoch(t *testing.T) {
	s:=adaptiveLifecycleFixture(t,func(context.Context,adapter.Outbound)error{return nil})
	resolver:=AdaptiveDNSResolver{IP:netip.MustParseAddr("1.1.1.1"),ServerName:"one.one.one.one"}
	if _,_,_,err:=s.beginTunnelDiagnostic(context.Background(),resolver,false);err==nil {t.Fatal("missing consent accepted")}
	bad:=resolver
	bad.IP=netip.MustParseAddr("127.0.0.1")
	if _,_,_,err:=s.beginTunnelDiagnostic(context.Background(),bad,true);err==nil {t.Fatal("private resolver accepted")}
	ctx,p,finish,err:=s.beginTunnelDiagnostic(context.Background(),resolver,true)
	if err!=nil {t.Fatal(err)}
	defer finish()
	if p.Generation()!=1||p.Scope()!=ac.ScopeTunnel||p.outbound.Tag()!="a" {t.Fatal("wrong path")}
	if _,err=s.RunAdaptiveCycle(context.Background());err==nil {t.Fatal("competing cycle allowed")}
	if _,err=p.DialContext(ctx,"tcp","127.0.0.1:443");err==nil {t.Fatal("private target accepted")}
	if _,err=p.DialContext(ctx,"tcp","owned.example:443");err==nil {t.Fatal("unpinned name accepted")}
	s.InterfaceUpdated(context.Background())
	if p.Generation()!=0||ctx.Err()==nil {t.Fatal("old path not invalidated")}
	if _,err=p.DialContext(ctx,"tcp","1.1.1.1:443");err==nil {t.Fatal("old path allowed")}
}
